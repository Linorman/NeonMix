cmake_minimum_required(VERSION 3.20)
project(neonmix_airplay_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(WIN32 AND MSVC)
  message(FATAL_ERROR "The pinned UxPlay protocol requires the MinGW pthread toolchain; the Rust Hub may remain MSVC-built")
endif()
find_package(Threads REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(NEONMIX_AIRPLAY_SOURCE_ROOT "${ROOT}/.local/airplay" CACHE PATH "Prepared pinned AirPlay sources")
set(NEONMIX_AUDIO_PLUGIN_DIR "${NEONMIX_AIRPLAY_SOURCE_ROOT}/plugins" CACHE PATH "Audio-only GStreamer plugin directory")
set(NEONMIX_AUDIO_REGISTRY "${NEONMIX_AIRPLAY_SOURCE_ROOT}/registry.bin" CACHE FILEPATH "Private GStreamer registry")
if(APPLE)
  set(DEFAULT_GST "${ROOT}/.local/gstreamer/prefix")
  set(DEFAULT_SDK "${NEONMIX_AIRPLAY_SOURCE_ROOT}/sdk")
else()
  set(DEFAULT_GST "")
  set(DEFAULT_SDK "")
endif()
set(NEONMIX_GSTREAMER_PREFIX "${DEFAULT_GST}" CACHE PATH "Extracted project-local GStreamer runtime/development prefix; empty uses pkg-config")
set(NEONMIX_GSTREAMER_SDK "${DEFAULT_SDK}" CACHE PATH "Separate extracted GStreamer headers; empty uses runtime prefix")
set(UP "${NEONMIX_AIRPLAY_SOURCE_ROOT}/patched/lib")
set(PLIST "${NEONMIX_AIRPLAY_SOURCE_ROOT}/src/libplist-2.6.0")
set(GST "${NEONMIX_GSTREAMER_PREFIX}")
set(SDK "${NEONMIX_GSTREAMER_SDK}")
if(NOT SDK)
  set(SDK "${GST}")
endif()
if(NOT EXISTS "${UP}/raop.c" OR NOT EXISTS "${PLIST}/src/plist.c")
  message(FATAL_ERROR "Prepare checksum-locked sources with tools/prepare_airplay.py --sources-only first")
endif()
file(GLOB PLAYFAIR "${UP}/playfair/*.c")
file(GLOB LLHTTP "${UP}/llhttp/*.c")
set(PROTOCOL byteutils compat crypto fairplay_playfair http_request http_response httpd logger netutils pairing raop raop_buffer raop_ntp raop_rtp srp utils)
foreach(NAME IN LISTS PROTOCOL)
  list(APPEND SOURCES "${UP}/${NAME}.c")
endforeach()
add_library(protocol STATIC ${SOURCES} ${PLAYFAIR} ${LLHTTP} disabled_media.c speaker_profile.c)
target_include_directories(protocol PUBLIC "${UP}" "${UP}/playfair" "${UP}/llhttp" "${PLIST}/include")
if(SDK)
  target_include_directories(protocol PUBLIC "${SDK}/include")
endif()
target_compile_definitions(protocol PRIVATE PLIST_210 PLIST_230 LIBPLIST_STATIC OPENSSL_API_COMPAT=0x10101000L)
file(GLOB PLIST_SOURCES "${PLIST}/src/*.c" "${PLIST}/libcnary/*.c")
# Match source filenames only: checkout directories may contain "test".
list(FILTER PLIST_SOURCES EXCLUDE REGEX "/(test[^/]*|plistutil|cnary)[.]c$")
add_library(plist STATIC ${PLIST_SOURCES})
# These archives are also linked into the shared protocol probe on ELF hosts.
set_target_properties(protocol plist PROPERTIES POSITION_INDEPENDENT_CODE ON)
include(CheckSymbolExists)
include(CheckStructHasMember)
if(NOT WIN32)
  set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE)
endif()
foreach(FEATURE strndup strptime timegm gmtime_r localtime_r)
  string(TOUPPER "${FEATURE}" UPPER_FEATURE)
  check_symbol_exists(${FEATURE} "string.h;time.h" HAVE_${UPPER_FEATURE})
endforeach()
check_struct_has_member("struct tm" tm_gmtoff "time.h" HAVE_TM_TM_GMTOFF)
check_struct_has_member("struct tm" tm_zone "time.h" HAVE_TM_TM_ZONE)
unset(CMAKE_REQUIRED_DEFINITIONS)
include(TestBigEndian)
test_big_endian(NEONMIX_BIG_ENDIAN)
if(NOT NEONMIX_BIG_ENDIAN)
  set(__LITTLE_ENDIAN__ 1)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  set(HAVE_ATTRIBUTE_CONSTRUCTOR 1)
endif()
configure_file(plist-config.h.in "${CMAKE_CURRENT_BINARY_DIR}/plist-config/config.h")
target_include_directories(plist PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/plist-config" "${PLIST}/include" "${PLIST}/src" "${PLIST}/libcnary/include" "${PLIST}/libcnary")
target_compile_definitions(plist PRIVATE HAVE_CONFIG_H PUBLIC LIBPLIST_STATIC)
if(NOT WIN32)
  target_compile_definitions(plist PRIVATE _GNU_SOURCE)
endif()
if(APPLE)
  # Use the pinned, extracted SDK/runtime; no framework or package installation.
  set(AUDIO_INCLUDES "${SDK}/include/gstreamer-1.0" "${SDK}/include/glib-2.0" "${SDK}/lib/glib-2.0/include" "${SDK}/lib/${CMAKE_LIBRARY_ARCHITECTURE}/glib-2.0/include")
  set(AUDIO_LIBRARIES "${GST}/lib/libgstreamer-1.0.dylib" "${GST}/lib/libgstapp-1.0.dylib" "${GST}/lib/libgobject-2.0.dylib" "${GST}/lib/libglib-2.0.dylib")
  set(CRYPTO_LIBRARY "${GST}/lib/libcrypto.dylib")
elseif(GST)
  set(AUDIO_INCLUDES "${SDK}/include/gstreamer-1.0" "${SDK}/include/glib-2.0" "${SDK}/lib/glib-2.0/include" "${SDK}/lib/${CMAKE_LIBRARY_ARCHITECTURE}/glib-2.0/include")
  foreach(LIBRARY gstreamer-1.0 gstapp-1.0 gobject-2.0 glib-2.0)
    string(MAKE_C_IDENTIFIER "${LIBRARY}" IDENTIFIER)
    find_library(AUDIO_${IDENTIFIER} NAMES ${LIBRARY} PATHS "${GST}/lib" "${GST}/lib/${CMAKE_LIBRARY_ARCHITECTURE}" NO_DEFAULT_PATH REQUIRED)
    list(APPEND AUDIO_LIBRARIES "${AUDIO_${IDENTIFIER}}")
  endforeach()
  find_library(CRYPTO_LIBRARY NAMES crypto libcrypto PATHS "${GST}/lib" "${GST}/lib/${CMAKE_LIBRARY_ARCHITECTURE}" NO_DEFAULT_PATH REQUIRED)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(AUDIO REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.24 gstreamer-app-1.0>=1.24)
  find_package(OpenSSL 1.1.1 REQUIRED COMPONENTS Crypto)
  set(AUDIO_LIBRARIES PkgConfig::AUDIO)
  set(CRYPTO_LIBRARY OpenSSL::Crypto)
endif()
if(WIN32)
  target_compile_definitions(protocol PUBLIC WIN32 _WIN32 NOMINMAX _WIN32_WINNT=0x0602)
  target_compile_definitions(plist PRIVATE WIN32 _WIN32_WINNT=0x0602)
  target_link_libraries(protocol PUBLIC ws2_32 iphlpapi)
endif()
target_link_libraries(protocol PUBLIC Threads::Threads ${CRYPTO_LIBRARY})
function(configure_audio_target target)
  target_include_directories(${target} PRIVATE ${AUDIO_INCLUDES})
  target_compile_definitions(${target} PRIVATE NEONMIX_AUDIO_PLUGIN_DIR="${NEONMIX_AUDIO_PLUGIN_DIR}" NEONMIX_AUDIO_REGISTRY="${NEONMIX_AUDIO_REGISTRY}")
  target_link_libraries(${target} PRIVATE protocol plist ${AUDIO_LIBRARIES})
  if(WIN32)
    target_link_libraries(${target} PRIVATE advapi32)
  endif()
  if(APPLE)
    set_target_properties(${target} PROPERTIES BUILD_RPATH "${GST}/lib")
  endif()
endfunction()
add_executable(neonmix-airplay-worker main.cpp)
configure_audio_target(neonmix-airplay-worker)
if(WIN32)
  add_executable(neonmix-airplay-pipe-probe EXCLUDE_FROM_ALL pipe_probe.cpp)
  target_compile_definitions(neonmix-airplay-pipe-probe PRIVATE _WIN32_WINNT=0x0602)
  target_link_libraries(neonmix-airplay-pipe-probe PRIVATE Threads::Threads advapi32 ws2_32)
endif()
add_executable(neonmix-airplay-audio-probe EXCLUDE_FROM_ALL audio_probe.cpp)
configure_audio_target(neonmix-airplay-audio-probe)
add_executable(neonmix-airplay-session-probe EXCLUDE_FROM_ALL session_probe.cpp)
configure_audio_target(neonmix-airplay-session-probe)

add_library(neonmix-airplay-probe-crypto SHARED EXCLUDE_FROM_ALL crypto_probe.c "${UP}/crypto.c")
target_include_directories(neonmix-airplay-probe-crypto PRIVATE "${UP}")
target_link_libraries(neonmix-airplay-probe-crypto PRIVATE protocol plist ${CRYPTO_LIBRARY})
set_target_properties(neonmix-airplay-probe-crypto PROPERTIES BUILD_RPATH "${GST}/lib")

# Exercises the same persisted-identity and pairing-init path as worker startup.
add_executable(neonmix-airplay-identity-probe EXCLUDE_FROM_ALL identity_probe.cpp)
target_compile_definitions(neonmix-airplay-identity-probe PRIVATE NEONMIX_IDENTITY_PROBE)
target_link_libraries(neonmix-airplay-identity-probe PRIVATE protocol plist)
if(WIN32)
  target_link_libraries(neonmix-airplay-identity-probe PRIVATE advapi32 ws2_32)
  target_link_options(neonmix-airplay-identity-probe PRIVATE -municode)
endif()
if(APPLE)
  set_target_properties(neonmix-airplay-identity-probe PROPERTIES BUILD_RPATH "${GST}/lib")
endif()
